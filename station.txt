local name = ... or "Radio 1"
peripheral.find("modem", rednet.open)
local mon = peripheral.find("monitor")
local CHUNK = 6 * 1024
local DUR = CHUNK * 8 / 48000
local skip, paused = false, false
local current = "-"
local volume = 3
local BAR_X = 6
local TOP = 6

local playlist = {}
for line in io.lines("playlist.txt") do
  local title, src = line:match("^(.-)|(.+)$")
  if title then table.insert(playlist, { title = title, src = src }) end
end

local function shuffle(t)
  for i = #t, 2, -1 do
    local j = math.random(i)
    t[i], t[j] = t[j], t[i]
  end
end

local function open(src)
  if src:match("^https?://") then
    return http.get{ url = src, binary = true }
  end
  return fs.open(src, "rb")
end

local function drawMon()
  if not mon then return end
  mon.setTextScale(1)
  local w, h = mon.getSize()
  local half = math.floor(w / 2)
  mon.setBackgroundColor(colors.black)
  mon.clear()
  mon.setTextColor(colors.white)
  mon.setCursorPos(2, 1) mon.write(name)
  mon.setCursorPos(2, 2) mon.write(current:sub(1, w - 2))
  mon.setCursorPos(2, 3)
  mon.setTextColor(paused and colors.red or colors.lime)
  mon.write(paused and "PAUSED" or "PLAYING")

  -- volume slider
  local barLen = w - BAR_X
  local filled = math.floor(volume / 3 * barLen + 0.5)
  mon.setTextColor(colors.white)
  mon.setCursorPos(2, 4) mon.write("Vol")
  mon.setCursorPos(BAR_X, 4)
  mon.setBackgroundColor(colors.lime)
  mon.write(string.rep(" ", filled))
  mon.setBackgroundColor(colors.gray)
  mon.write(string.rep(" ", barLen - filled))
  mon.setBackgroundColor(colors.black)

  -- buttons
  local left = paused and colors.green or colors.orange
  for y = TOP, h do
    mon.setCursorPos(1, y)
    mon.setBackgroundColor(left)
    mon.write(string.rep(" ", half))
    mon.setBackgroundColor(colors.blue)
    mon.write(string.rep(" ", w - half))
  end
  local mid = math.floor((TOP + h) / 2)
  local label = paused and "PLAY" or "PAUSE"
  mon.setBackgroundColor(left)
  mon.setCursorPos(math.floor((half - #label) / 2) + 1, mid)
  mon.write(label)
  mon.setBackgroundColor(colors.blue)
  mon.setCursorPos(half + math.floor((w - half - 4) / 2) + 1, mid)
  mon.write("SKIP")
end

local function setPaused(state)
  paused = state
  print(paused and "Paused" or "Resumed")
  drawMon()
end

local function setVolume(v)
  volume = math.max(0, math.min(3, math.floor(v * 4 + 0.5) / 4))
  print("Volume: " .. volume)
  drawMon()
end

local function doSkip()
  skip = true
  paused = false
end

local function broadcast()
  while true do
    shuffle(playlist)
    for _, song in ipairs(playlist) do
      current = song.title
      print("Now playing: " .. current)
      drawMon()
      local h = open(song.src)
      if h then
        skip = false
        local t = os.clock()
        while not skip do
          if paused then
            while paused and not skip do sleep(0.2) end
            t = os.clock()
          end
          if skip then break end
          local chunk = h.read(CHUNK)
          if not chunk then break end
          rednet.broadcast({ station = name, title = current, data = chunk, volume = volume }, "radio")
          t = t + DUR
          sleep(math.max(0, t - os.clock() - 0.1))
        end
        h.close()
      else
        print("Can't open " .. song.title)
      end
    end
  end
end

local function control()
  while true do
    local ev, a, b, c = os.pullEvent()
    if ev == "key" and a == keys.s then
      doSkip()
    elseif ev == "key" and a == keys.p then
      setPaused(not paused)
    elseif ev == "char" and (a == "+" or a == "=") then
      setVolume(volume + 0.25)
    elseif ev == "char" and a == "-" then
      setVolume(volume - 0.25)
    elseif ev == "monitor_touch" and mon then
      local w = mon.getSize()
      if c == 4 and b >= BAR_X then
        setVolume((b - BAR_X + 1) / (w - BAR_X) * 3)
      elseif c >= TOP then
        if b <= math.floor(w / 2) then setPaused(not paused) else doSkip() end
      end
    elseif ev == "rednet_message" and c == "radio_ctrl" and type(b) == "table" then
      if b.skip == name then doSkip() end
      if b.pause == name then setPaused(b.state) end
    end
  end
end

print("Broadcasting as " .. name .. "  (S skip, P pause, +/- volume)")
parallel.waitForAny(broadcast, control)
