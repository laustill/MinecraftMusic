peripheral.find("modem", rednet.open)
local speaker = peripheral.find("speaker")
local dfpwm = require("cc.audio.dfpwm")
local dec = dfpwm.make_decoder()
local list, known = {}, {}
local tuned, current

local function show()
  term.clear() term.setCursorPos(1, 1)
  print("Station: " .. (tuned or "searching..."))
  print("Playing: " .. (current or "-"))
  print("\n<- -> station   S skip")
end

local function tune(name)
  if name == tuned then return end
  tuned, current = name, nil
  dec = dfpwm.make_decoder()
  speaker.stop()
  show()
end

local function listen()
  while true do
    local id, msg = rednet.receive("radio")
    if type(msg) == "table" and msg.data and msg.station then
      if not known[msg.station] then
        known[msg.station] = true
        table.insert(list, msg.station)
        if not tuned then tune(msg.station) end
      end
      if msg.station == tuned then
        if msg.title ~= current then current = msg.title show() end
        local buf = dec(msg.data)
        while not speaker.playAudio(buf, msg.volume or 3) do
          os.pullEvent("speaker_audio_empty")
        end
      end
    end
  end
end

local function remote()
  while true do
    local id, msg = rednet.receive("radio_ctrl")
    if type(msg) == "table" and msg.tune then
      tune(msg.tune)
    end
  end
end

local function controls()
  while true do
    local _, key = os.pullEvent("key")
    if (key == keys.left or key == keys.right) and #list > 1 then
      local i = 1
      for n, s in ipairs(list) do if s == tuned then i = n end end
      i = key == keys.right and i % #list + 1 or (i - 2) % #list + 1
      tune(list[i])
    elseif key == keys.s and tuned then
      rednet.broadcast({ skip = tuned }, "radio_ctrl")
      speaker.stop()
    end
  end
end

show()
parallel.waitForAny(listen, remote, controls)
