peripheral.find("modem", rednet.open)
local list, known, sel = {}, {}, 1

local function draw()
  term.clear() term.setCursorPos(1, 1)
  print("=== ALL RADIOS ===")
  if #list == 0 then print("Searching...") end
  for i, s in ipairs(list) do
    print((i == sel and "> " or "  ") .. s)
  end
  print("\nUp/Down select")
  print("Enter = tune all")
end

local function listen()
  while true do
    local _, msg = rednet.receive("radio")
    if type(msg) == "table" and msg.station and not known[msg.station] then
      known[msg.station] = true
      table.insert(list, msg.station)
      draw()
    end
  end
end

local function controls()
  while true do
    local _, key = os.pullEvent("key")
    if key == keys.up and sel > 1 then sel = sel - 1 draw()
    elseif key == keys.down and sel < #list then sel = sel + 1 draw()
    elseif key == keys.enter and list[sel] then
      rednet.broadcast({ tune = list[sel] }, "radio_ctrl")
      draw()
      print("Tuned all to " .. list[sel])
    end
  end
end

draw()
parallel.waitForAny(listen, controls)
